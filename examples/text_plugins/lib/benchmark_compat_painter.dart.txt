Widget wrapScope({
  required List<TextPlugin> plugins,
  required Variant variant,
  required Widget child,
}) {
  return TextPluginScope.multiple(plugins: plugins, child: child);
}

SearchInPagePlugin makeSearch(Variant variant) => SearchInPagePlugin();

/// Highlights one word of a text and moves on with [step].
class KaraokePlugin extends TextPlugin {
  static final RegExp _word = RegExp(r'\S+');

  TextDelegate? _delegate;
  List<TextRange> _words = const <TextRange>[];
  int _index = 0;

  @override
  void didAddText(TextDelegate delegate) {
    _delegate = delegate;
    _words = <TextRange>[
      for (final Match match in _word.allMatches(delegate.text))
        TextRange(start: match.start, end: match.end),
    ];
    delegate.backgroundPainter = _KaraokePainter(_words[_index]);
  }

  void step() {
    _index = (_index + 1) % _words.length;
    _delegate!.backgroundPainter = _KaraokePainter(_words[_index]);
  }
}

class _KaraokePainter extends TextPluginPainter {
  _KaraokePainter(this.range);

  final TextRange range;

  @override
  void paint(Canvas canvas, TextLayout layout) {
    final paint = Paint()..color = const Color(0x5500ACC1);
    for (final TextBox box in layout.getBoxesForSelection(
      TextSelection(baseOffset: range.start, extentOffset: range.end),
    )) {
      canvas.drawRect(box.toRect(), paint);
      canvas.drawLine(
        Offset(box.left, box.bottom),
        Offset(box.right, box.bottom),
        paint,
      );
    }
  }

  @override
  bool shouldRepaint(_KaraokePainter oldPainter) => oldPainter.range != range;
}
